import React, { useEffect, useState } from 'react';
import {
  View,
  Text,
  StyleSheet,
  ScrollView,
  TouchableOpacity,
  RefreshControl,
} from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { router } from 'expo-router';
import { Ionicons } from '@expo/vector-icons';
import { LinearGradient } from 'expo-linear-gradient';
import { theme } from '../../src/theme';
import { useAuthStore } from '../../src/stores/authStore';
import { supabase } from '../../src/lib/supabase';

const COMMISSION_RATE = 0.15; // 15% commission on delivery fees

interface WeeklyEarnings {
  weekStart: string;
  weekEnd: string;
  totalDeliveryFees: number;
  commission: number;
  deliveries: number;
  payoutStatus?: string;
}

export default function DriverEarnings() {
  const { user } = useAuthStore();
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [weeklyEarnings, setWeeklyEarnings] = useState<WeeklyEarnings[]>([]);
  const [totalEarned, setTotalEarned] = useState(0);
  const [currentWeekEarnings, setCurrentWeekEarnings] = useState(0);
  const [pendingPayout, setPendingPayout] = useState(0);

  useEffect(() => {
    if (user) {
      fetchEarnings();
      fetchDriverStats();
    }
  }, [user]);

  const fetchDriverStats = async () => {
    if (!user) return;
    
    // Get driver's total stats
    const { data } = await supabase
      .from('drivers')
      .select('total_earned')
      .eq('id', user.id)
      .maybeSingle();
    
    if (data) {
      // Update local state if needed
    }
  };

  const fetchEarnings = async () => {
    if (!user) return;
    
    setLoading(true);
    
    // Get all delivered orders with commission
    const { data: orders } = await supabase
      .from('orders')
      .select('delivery_fee, delivered_at, driver_commission')
      .eq('driver_id', user.id)
      .eq('status', 'delivered')
      .order('delivered_at', { ascending: false });
    
    if (orders && orders.length > 0) {
      // Group by week using driver_commission (already calculated by trigger)
      const weeks: { [key: string]: { totalFees: number; commission: number; count: number } } = {};
      
      orders.forEach(order => {
        const date = new Date(order.delivered_at);
        const weekStart = getWeekStart(date);
        const weekKey = weekStart.toISOString().split('T')[0];
        
        if (!weeks[weekKey]) {
          weeks[weekKey] = { totalFees: 0, commission: 0, count: 0 };
        }
        weeks[weekKey].totalFees += order.delivery_fee;
        weeks[weekKey].commission += order.driver_commission || (order.delivery_fee * COMMISSION_RATE);
        weeks[weekKey].count += 1;
      });
      
      // Get payout status from driver_weekly_earnings table
      const { data: payouts } = await supabase
        .from('driver_weekly_earnings')
        .select('week_start, payout_status')
        .eq('driver_id', user.id);
      
      const payoutMap: { [key: string]: string } = {};
      if (payouts) {
        payouts.forEach(p => {
          payoutMap[p.week_start] = p.payout_status;
        });
      }
      
      // Calculate commission and format
      const weekData: WeeklyEarnings[] = Object.entries(weeks).map(([weekStart, data]) => {
        const startDate = new Date(weekStart);
        const endDate = new Date(startDate);
        endDate.setDate(endDate.getDate() + 6);
        
        return {
          weekStart,
          weekEnd: endDate.toISOString().split('T')[0],
          totalDeliveryFees: data.totalFees,
          commission: Math.round(data.commission),
          deliveries: data.count,
          payoutStatus: payoutMap[weekStart] || 'pending',
        };
      });
      
      setWeeklyEarnings(weekData);
      
      // Calculate total earned
      const total = weekData.reduce((sum, week) => sum + week.commission, 0);
      setTotalEarned(total);
      
      // Calculate pending payout (unpaid weeks)
      const pending = weekData
        .filter(w => w.payoutStatus === 'pending')
        .reduce((sum, w) => sum + w.commission, 0);
      setPendingPayout(pending);
      
      // Current week earnings
      const currentWeek = getWeekStart(new Date());
      const currentWeekKey = currentWeek.toISOString().split('T')[0];
      const currentWeekData = weekData.find(w => w.weekStart === currentWeekKey);
      setCurrentWeekEarnings(currentWeekData?.commission || 0);
    }
    
    setLoading(false);
  };

  const getWeekStart = (date: Date): Date => {
    const d = new Date(date);
    const day = d.getDay();
    const diff = d.getDate() - day + (day === 0 ? -6 : 1);
    return new Date(d.setDate(diff));
  };

  const formatWeekRange = (weekStart: string, weekEnd: string) => {
    const start = new Date(weekStart);
    const end = new Date(weekEnd);
    return `${start.toLocaleDateString()} - ${end.toLocaleDateString()}`;
  };

  const getPayoutStatusColor = (status?: string) => {
    switch (status) {
      case 'completed': return '#10b981';
      case 'processing': return '#3b82f6';
      default: return '#f59e0b';
    }
  };

  const onRefresh = async () => {
    setRefreshing(true);
    await fetchEarnings();
    await fetchDriverStats();
    setRefreshing(false);
  };

  return (
    <SafeAreaView style={styles.container}>
      <View style={styles.header}>
        <TouchableOpacity onPress={() => router.back()} style={styles.backButton}>
          <Ionicons name="arrow-back" size={24} color={theme.colors.textDark} />
        </TouchableOpacity>
        <Text style={styles.title}>Earnings</Text>
        <View style={{ width: 40 }} />
      </View>

      <ScrollView
        showsVerticalScrollIndicator={false}
        refreshControl={<RefreshControl refreshing={refreshing} onRefresh={onRefresh} />}
      >
        {/* Current Week Card */}
        <LinearGradient
          colors={theme.gradients.gold}
          style={styles.currentCard}
        >
          <Text style={styles.currentLabel}>This week's earnings</Text>
          <Text style={styles.currentValue}>{currentWeekEarnings} MAD</Text>
          <View style={styles.commissionBadge}>
            <Text style={styles.commissionBadgeText}>15% commission</Text>
          </View>
          <Text style={styles.currentNote}>
            Based on {Math.round(currentWeekEarnings / 2.25)} deliveries this week
          </Text>
        </LinearGradient>

        {/* Stats Row */}
        <View style={styles.statsRow}>
          <View style={styles.statCard}>
            <Text style={styles.statValue}>{totalEarned} MAD</Text>
            <Text style={styles.statLabel}>Total earned</Text>
          </View>
          <View style={styles.statDivider} />
          <View style={styles.statCard}>
            <Text style={styles.statValue}>{pendingPayout} MAD</Text>
            <Text style={styles.statLabel}>Pending payout</Text>
          </View>
        </View>

        {/* Info Card - How earnings work */}
        <View style={styles.infoCard}>
          <Ionicons name="information-circle-outline" size={24} color={theme.colors.accent} />
          <View style={styles.infoContent}>
            <Text style={styles.infoTitle}>How earnings work</Text>
            <Text style={styles.infoText}>
              • You earn 15% commission on delivery fees
              • Delivery fee is 15 MAD per order
              • You earn 2.25 MAD per delivery
              • Earnings are calculated weekly
              • Paid every Monday via bank transfer
            </Text>
          </View>
        </View>

        {/* Weekly Breakdown */}
        <View style={styles.weeklySection}>
          <Text style={styles.sectionTitle}>Weekly breakdown</Text>
          {weeklyEarnings.length === 0 ? (
            <View style={styles.emptyState}>
              <Ionicons name="calendar-outline" size={48} color={theme.colors.textLight} />
              <Text style={styles.emptyText}>No earnings yet</Text>
              <Text style={styles.emptySubtext}>Complete deliveries to see earnings</Text>
            </View>
          ) : (
            weeklyEarnings.map((week, index) => (
              <View key={index} style={styles.weekCard}>
                <View style={styles.weekHeader}>
                  <Text style={styles.weekRange}>
                    {formatWeekRange(week.weekStart, week.weekEnd)}
                  </Text>
                  <View style={[styles.payoutBadge, { backgroundColor: getPayoutStatusColor(week.payoutStatus) + '20' }]}>
                    <Text style={[styles.payoutText, { color: getPayoutStatusColor(week.payoutStatus) }]}>
                      {week.payoutStatus?.toUpperCase() || 'PENDING'}
                    </Text>
                  </View>
                </View>
                <View style={styles.weekStats}>
                  <View style={styles.weekStat}>
                    <Text style={styles.weekStatValue}>{week.deliveries}</Text>
                    <Text style={styles.weekStatLabel}>Deliveries</Text>
                  </View>
                  <View style={styles.weekStatDivider} />
                  <View style={styles.weekStat}>
                    <Text style={styles.weekStatValue}>{week.totalDeliveryFees} MAD</Text>
                    <Text style={styles.weekStatLabel}>Total fees</Text>
                  </View>
                  <View style={styles.weekStatDivider} />
                  <View style={styles.weekStat}>
                    <Text style={styles.weekStatValue}>{week.commission} MAD</Text>
                    <Text style={styles.weekStatLabel}>Your earnings</Text>
                  </View>
                </View>
                <View style={styles.progressBar}>
                  <View style={[styles.progressFill, { width: week.payoutStatus === 'completed' ? '100%' : '0%' }]} />
                </View>
                {week.payoutStatus === 'completed' && (
                  <Text style={styles.paidLabel}>✓ Paid</Text>
                )}
              </View>
            ))
          )}
        </View>
      </ScrollView>
    </SafeAreaView>
  );
}

import React, { useEffect, useState } from 'react';
import {
  View,
  Text,
  StyleSheet,
  ScrollView,
  TouchableOpacity,
  RefreshControl,
} from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { router } from 'expo-router';
import { Ionicons } from '@expo/vector-icons';
import { theme } from '../../src/theme';
import { useAuthStore } from '../../src/stores/authStore';
import { supabase } from '../../src/lib/supabase';

interface DeliveryHistory {
  id: string;
  order_number: string;
  delivery_address: string;
  delivery_fee: number;
  status: string;
  delivered_at: string | null;
}

export default function DriverHistory() {
  const { user } = useAuthStore();
  const [deliveries, setDeliveries] = useState<DeliveryHistory[]>([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [filter, setFilter] = useState<'all' | 'completed' | 'cancelled'>('all');

  useEffect(() => {
    if (user) {
      fetchHistory();
    }
  }, [user, filter]);

  const fetchHistory = async () => {
    if (!user) return;
    
    setLoading(true);
    
    let query = supabase
      .from('orders')
      .select('id, order_number, delivery_address, delivery_fee, status, delivered_at')
      .eq('driver_id', user.id);
    
    if (filter === 'completed') {
      query = query.eq('status', 'delivered');
    } else if (filter === 'cancelled') {
      query = query.in('status', ['cancelled', 'rejected']);
    }
    
    const { data } = await query.order('created_at', { ascending: false });
    
    if (data) {
      setDeliveries(data);
    }
    
    setLoading(false);
  };

  const onRefresh = async () => {
    setRefreshing(true);
    await fetchHistory();
    setRefreshing(false);
  };

  const getStatusColor = (status: string) => {
    switch (status) {
      case 'delivered': return theme.colors.success;
      case 'cancelled': return theme.colors.error;
      default: return theme.colors.warning;
    }
  };

  const getStatusIcon = (status: string) => {
    switch (status) {
      case 'delivered': return 'checkmark-circle';
      case 'cancelled': return 'close-circle';
      default: return 'time-outline';
    }
  };

  const formatDate = (dateString: string | null) => {
    if (!dateString) return 'Pending';
    const date = new Date(dateString);
    return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
  };

  return (
    <SafeAreaView style={styles.container}>
      <View style={styles.header}>
        <TouchableOpacity onPress={() => router.back()} style={styles.backButton}>
          <Ionicons name="arrow-back" size={24} color={theme.colors.textDark} />
        </TouchableOpacity>
        <Text style={styles.title}>Delivery History</Text>
        <View style={{ width: 40 }} />
      </View>

      {/* Filter Tabs */}
      <View style={styles.filterContainer}>
        <TouchableOpacity
          style={[styles.filterTab, filter === 'all' && styles.filterActive]}
          onPress={() => setFilter('all')}
        >
          <Text style={[styles.filterText, filter === 'all' && styles.filterTextActive]}>All</Text>
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.filterTab, filter === 'completed' && styles.filterActive]}
          onPress={() => setFilter('completed')}
        >
          <Text style={[styles.filterText, filter === 'completed' && styles.filterTextActive]}>Completed</Text>
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.filterTab, filter === 'cancelled' && styles.filterActive]}
          onPress={() => setFilter('cancelled')}
        >
          <Text style={[styles.filterText, filter === 'cancelled' && styles.filterTextActive]}>Cancelled</Text>
        </TouchableOpacity>
      </View>

      <ScrollView
        showsVerticalScrollIndicator={false}
        refreshControl={<RefreshControl refreshing={refreshing} onRefresh={onRefresh} />}
        contentContainerStyle={styles.content}
      >
        {deliveries.length === 0 ? (
          <View style={styles.emptyState}>
            <Ionicons name="time-outline" size={64} color={theme.colors.textLight} />
            <Text style={styles.emptyTitle}>No deliveries yet</Text>
            <Text style={styles.emptyText}>Your delivery history will appear here</Text>
          </View>
        ) : (
          deliveries.map((delivery) => (
            <TouchableOpacity
              key={delivery.id}
              style={styles.deliveryCard}
              onPress={() => router.push(`/(driver)/orders/${delivery.id}`)}
            >
              <View style={styles.deliveryHeader}>
                <View style={styles.deliveryInfo}>
                  <Text style={styles.deliveryNumber}>{delivery.order_number}</Text>
                  <Text style={styles.deliveryAddress} numberOfLines={1}>{delivery.delivery_address}</Text>
                </View>
                <View style={[styles.statusBadge, { backgroundColor: getStatusColor(delivery.status) + '20' }]}>
                  <Ionicons name={getStatusIcon(delivery.status)} size={12} color={getStatusColor(delivery.status)} />
                  <Text style={[styles.statusText, { color: getStatusColor(delivery.status) }]}>
                    {delivery.status.toUpperCase()}
                  </Text>
                </View>
              </View>
              <View style={styles.deliveryFooter}>
                <Text style={styles.deliveryDate}>{formatDate(delivery.delivered_at)}</Text>
                <Text style={styles.deliveryFee}>+{delivery.delivery_fee} MAD</Text>
              </View>
            </TouchableOpacity>
          ))
        )}
      </ScrollView>
    </SafeAreaView>
  );
}

import { Ionicons } from '@expo/vector-icons';
import { theme } from '../../src/theme';
import { Button } from '../../src/components/ui/Button';
import { useAuthStore } from '../../src/stores/authStore';
import { supabase } from '../../src/lib/supabase';
import * as Location from 'expo-location';

interface Order {
  id: string;
  order_number: string;
  delivery_address: string;
  total: number;
  delivery_fee: number;
  created_at: string;
}

export default function DriverDashboard() {
  const { user } = useAuthStore();
  const [isOnline, setIsOnline] = useState(false);
  const [isTrackingLocation, setIsTrackingLocation] = useState(false);
  const [availableOrders, setAvailableOrders] = useState<Order[]>([]);
  const [activeOrder, setActiveOrder] = useState<Order | null>(null);
  const [earnings, setEarnings] = useState({ today: 0, week: 0, total: 0 });
  const [refreshing, setRefreshing] = useState(false);
  const [locationSubscription, setLocationSubscription] = useState<any>(null);

  useEffect(() => {
    if (user) {
      fetchDriverStatus();
      fetchAvailableOrders();
      fetchActiveOrder();
      fetchEarnings();
    }
  }, [user]);

  useEffect(() => {
    const subscription = supabase
      .channel('driver-orders')
      .on('postgres_changes', 
        { event: 'INSERT', schema: 'public', table: 'orders', filter: 'driver_id=is.null' },
        () => fetchAvailableOrders()
      )
      .subscribe();

    return () => {
      subscription.unsubscribe();
    };
  }, []);

  // Clean up location tracking on unmount
  useEffect(() => {
    return () => {
      if (locationSubscription) {
        locationSubscription.remove();
      }
    };
  }, [locationSubscription]);

  const requestLocationPermission = async () => {
    const { status } = await Location.requestForegroundPermissionsAsync();
    if (status !== 'granted') {
      Alert.alert('Permission needed', 'Location permission is required to share your location with customers');
      return false;
    }
    return true;
  };

  const startLocationTracking = async (driverId: string) => {
    const hasPermission = await requestLocationPermission();
    if (!hasPermission) return;

    // Get initial location
    const location = await Location.getCurrentPositionAsync({});
    await updateDriverLocation(driverId, location.coords.latitude, location.coords.longitude);

    // Start watching position
    const subscription = await Location.watchPositionAsync(
      {
        accuracy: Location.Accuracy.High,
        timeInterval: 5000,
        distanceInterval: 10,
      },
      async (newLocation) => {
        await updateDriverLocation(driverId, newLocation.coords.latitude, newLocation.coords.longitude);
      }
    );

    setLocationSubscription(subscription);
  };

  const updateDriverLocation = async (driverId: string, latitude: number, longitude: number) => {
    // Update drivers table
    await supabase
      .from('drivers')
      .update({
        current_latitude: latitude,
        current_longitude: longitude,
        last_location_update: new Date().toISOString(),
      })
      .eq('id', driverId);

    // Update active order if any
    if (activeOrder) {
      await supabase
        .from('orders')
        .update({
          driver_latitude: latitude,
          driver_longitude: longitude,
          last_location_update: new Date().toISOString(),
        })
        .eq('driver_id', driverId)
        .in('status', ['accepted', 'picked_up', 'on_the_way']);
    }
  };

  const stopLocationTracking = () => {
    if (locationSubscription) {
      locationSubscription.remove();
      setLocationSubscription(null);
    }
  };

const fetchDriverStatus = async () => {
  if (!user) return;
  
  // Use maybeSingle() and also handle the case of multiple rows
  const { data, error } = await supabase
    .from('drivers')
    .select('status')
    .eq('id', user.id)
    .maybeSingle();
  
  if (error) {
    console.log('Error fetching driver status:', error);
  }
  
  // If data is an array (multiple rows), take the first one
  const status = Array.isArray(data) ? data[0]?.status : data?.status;
  
  if (status) {
    setIsOnline(status === 'online');
  }
};

  const fetchAvailableOrders = async () => {
    const { data } = await supabase
      .from('orders')
      .select('id, order_number, delivery_address, total, delivery_fee, created_at')
      .eq('status', 'confirmed')
      .is('driver_id', null);
    
    if (data) {
      setAvailableOrders(data);
    }
  };

const fetchActiveOrder = async () => {
  if (!user) return;
  
  const { data } = await supabase
    .from('orders')
    .select('id, order_number, delivery_address, total, delivery_fee')
    .eq('driver_id', user.id)
    .in('status', ['accepted', 'picked_up', 'on_the_way'])
    .maybeSingle();  
  
  if (data) {
    setActiveOrder(data);
  } else {
    setActiveOrder(null);
  }
};

const fetchEarnings = async () => {
  if (!user) return;
  
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  
  const weekAgo = new Date();
  weekAgo.setDate(weekAgo.getDate() - 7);
  
  const { data } = await supabase
    .from('orders')
    .select('delivery_fee, delivered_at')
    .eq('driver_id', user.id)
    .eq('status', 'delivered');
  
  if (data && data.length > 0) {
    const todayEarnings = data
      .filter(o => new Date(o.delivered_at) >= today)
      .reduce((sum, o) => sum + o.delivery_fee, 0);
    
    const weekEarnings = data
      .filter(o => new Date(o.delivered_at) >= weekAgo)
      .reduce((sum, o) => sum + o.delivery_fee, 0);
    
    const totalEarnings = data.reduce((sum, o) => sum + o.delivery_fee, 0);
    
    setEarnings({ today: todayEarnings, week: weekEarnings, total: totalEarnings });
  } else {
    setEarnings({ today: 0, week: 0, total: 0 });
  }
};

  const onRefresh = async () => {
    setRefreshing(true);
    await fetchDriverStatus();
    await fetchAvailableOrders();
    await fetchActiveOrder();
    await fetchEarnings();
    setRefreshing(false);
  };

  const handleToggleStatus = async () => {
    if (!user) return;
    
    const newStatus = !isOnline;
    const { error } = await supabase
      .from('drivers')
      .update({ status: newStatus ? 'online' : 'offline' })
      .eq('id', user.id);
    
    if (!error) {
      setIsOnline(newStatus);
      
      if (newStatus) {
        // Start location tracking when going online
        await startLocationTracking(user.id);
        setIsTrackingLocation(true);
        Alert.alert('You are now ONLINE', 'Your location will be shared with customers');
      } else {
        // Stop location tracking when going offline
        stopLocationTracking();
        setIsTrackingLocation(false);
        Alert.alert('You are now OFFLINE', 'You will not receive new orders');
      }
    }
  };

  const handleAcceptOrder = async (orderId: string) => {
  if (!user) return;
  
  Alert.alert(
    'Accept Delivery',
    'Are you sure you want to accept this delivery?',
    [
      { text: 'Cancel', style: 'cancel' },
      {
        text: 'Accept',
        onPress: async () => {
          // Atomic update - only accept if driver_id is still null
          const { data, error } = await supabase
            .from('orders')
            .update({ 
              driver_id: user.id, 
              status: 'accepted', 
              accepted_at: new Date().toISOString() 
            })
            .eq('id', orderId)
            .is('driver_id', null)  // ✅ Only accept if no driver assigned yet
            .select();
          
          if (error) {
            Alert.alert('Error', 'Failed to accept order');
          } else if (data && data.length === 0) {
            Alert.alert('Order Taken', 'This order has already been accepted by another driver');
          } else {
            await fetchAvailableOrders();
            await fetchActiveOrder();
            Alert.alert('Order Accepted', 'Navigate to pick up the order');
          }
        },
      },
    ]
  );
};

  return (
    <SafeAreaView style={styles.container}>
      <ScrollView
        showsVerticalScrollIndicator={false}
        refreshControl={<RefreshControl refreshing={refreshing} onRefresh={onRefresh} />}
      >
        {/* Header */}
        <View style={styles.header}>
          <View>
            <Text style={styles.greeting}>Driver Dashboard</Text>
            <Text style={styles.name}>{user?.email?.split('@')[0] || 'Driver'}</Text>
          </View>
          <TouchableOpacity onPress={() => router.push('/(driver)/profile')}>
            <View style={styles.avatar}>
              <Ionicons name="person" size={24} color={theme.colors.textWhite} />
            </View>
          </TouchableOpacity>
        </View>

        {/* Status Card */}
        <View style={styles.statusCard}>
          <View>
            <Text style={styles.statusLabel}>Current Status</Text>
            <Text style={[styles.statusValue, isOnline && styles.statusOnline]}>
              {isOnline ? '🟢 ONLINE' : '⚫ OFFLINE'}
            </Text>
            {isTrackingLocation && (
              <Text style={styles.trackingText}>📍 Sharing location</Text>
            )}
          </View>
          <Button
            title={isOnline ? 'GO OFFLINE' : 'GO ONLINE'}
            variant={isOnline ? 'danger' : 'success'}
            onPress={handleToggleStatus}
            size="sm"
          />
        </View>

        {/* Active Order */}
        {activeOrder && (
          <View style={styles.activeSection}>
            <Text style={styles.sectionTitle}>ACTIVE DELIVERY</Text>
            <TouchableOpacity 
              style={styles.activeCard}
              onPress={() => router.push(`/(driver)/orders/${activeOrder.id}`)}
            >
              <Text style={styles.activeOrderNumber}>{activeOrder.order_number}</Text>
              <Text style={styles.activeAddress}>{activeOrder.delivery_address}</Text>
              <Text style={styles.activeFee}>+{activeOrder.delivery_fee} MAD</Text>
              <View style={styles.activeStatus}>
                <Text style={styles.activeStatusText}>Tap to navigate →</Text>
              </View>
            </TouchableOpacity>
          </View>
        )}

        {/* Earnings Preview */}
        <View style={styles.earningsSection}>
          <Text style={styles.sectionTitle}>TODAY'S EARNINGS</Text>
          <View style={styles.earningsCard}>
            <View style={styles.earningItem}>
              <Text style={styles.earningValue}>{earnings.today} MAD</Text>
              <Text style={styles.earningLabel}>Today</Text>
            </View>
            <View style={styles.earningDivider} />
            <View style={styles.earningItem}>
              <Text style={styles.earningValue}>{earnings.week} MAD</Text>
              <Text style={styles.earningLabel}>This Week</Text>
            </View>
            <View style={styles.earningDivider} />
            <View style={styles.earningItem}>
              <Text style={styles.earningValue}>{earnings.total} MAD</Text>
              <Text style={styles.earningLabel}>Total</Text>
            </View>
          </View>
          <TouchableOpacity onPress={() => router.push('/(driver)/earnings')}>
            <Text style={styles.earningsLink}>View Details →</Text>
          </TouchableOpacity>
        </View>

        {/* Available Orders */}
        {isOnline && (
          <View style={styles.ordersSection}>
            <Text style={styles.sectionTitle}>AVAILABLE ORDERS ({availableOrders.length})</Text>
            {availableOrders.length === 0 ? (
              <View style={styles.emptyState}>
                <Ionicons name="bicycle-outline" size={48} color={theme.colors.textLight} />
                <Text style={styles.emptyText}>No orders available</Text>
                <Text style={styles.emptySubtext}>Check back later</Text>
              </View>
            ) : (
              availableOrders.map((order) => (
                <View key={order.id} style={styles.orderCard}>
                  <View style={styles.orderHeader}>
                    <Text style={styles.orderNumber}>{order.order_number}</Text>
                    <Text style={styles.orderFee}>+{order.delivery_fee} MAD</Text>
                  </View>
                  <View style={styles.orderAddress}>
                    <Ionicons name="location-outline" size={16} color={theme.colors.textLight} />
                    <Text style={styles.addressText}>{order.delivery_address}</Text>
                  </View>
                  <View style={styles.orderFooter}>
                    <Text style={styles.orderTotal}>{order.total} MAD</Text>
                    <Button
                      title="ACCEPT"
                      variant="primary"
                      size="sm"
                      onPress={() => handleAcceptOrder(order.id)}
                    />
                  </View>
                </View>
              ))
            )}
          </View>
        )}
      </ScrollView>
    </SafeAreaView>
  );
}

import React, { useEffect, useState } from 'react';
import {
  View,
  Text,
  StyleSheet,
  ScrollView,
  TouchableOpacity,
  Alert,
} from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { router } from 'expo-router';
import { Ionicons } from '@expo/vector-icons';
import { theme } from '../../src/theme';
import { Button } from '../../src/components/ui/Button';
import { useAuthStore } from '../../src/stores/authStore';
import { supabase } from '../../src/lib/supabase';

export default function DriverProfile() {
  const { user, signOut } = useAuthStore();
  const [profile, setProfile] = useState<any>(null);
  const [stats, setStats] = useState({
    totalDeliveries: 0,
    totalEarnings: 0,
    rating: 0,
  });

  useEffect(() => {
    if (user) {
      fetchProfile();
      fetchStats();
    }
  }, [user]);

  const fetchProfile = async () => {
    if (!user) return;
    
    const { data } = await supabase
      .from('drivers')
      .select('*')
      .eq('id', user.id)
      .single();
    
    if (data) {
      setProfile(data);
    }
  };

  const fetchStats = async () => {
    if (!user) return;
    
    const { data: deliveries } = await supabase
      .from('orders')
      .select('delivery_fee')
      .eq('driver_id', user.id)
      .eq('status', 'delivered');
    
    if (deliveries) {
      setStats({
        totalDeliveries: deliveries.length,
        totalEarnings: deliveries.reduce((sum, d) => sum + d.delivery_fee, 0),
        rating: 4.9,
      });
    }
  };

  const handleSignOut = () => {
    Alert.alert(
      'Sign Out',
      'Are you sure you want to sign out?',
      [
        { text: 'Cancel', style: 'cancel' },
        { 
          text: 'Sign Out', 
          style: 'destructive',
          onPress: async () => {
            await signOut();
            router.replace('/(auth)/login');
          }
        },
      ]
    );
  };

  return (
    <SafeAreaView style={styles.container}>
      <ScrollView showsVerticalScrollIndicator={false}>
        {/* Header */}
        <View style={styles.header}>
          <TouchableOpacity onPress={() => router.back()} style={styles.backButton}>
            <Ionicons name="arrow-back" size={24} color={theme.colors.textDark} />
          </TouchableOpacity>
          <Text style={styles.title}>My Profile</Text>
          <View style={{ width: 40 }} />
        </View>

        {/* Profile Card */}
        <View style={styles.profileCard}>
          <View style={styles.avatar}>
            <Text style={styles.avatarText}>
              {profile?.first_name?.charAt(0) || user?.email?.charAt(0) || 'D'}
            </Text>
          </View>
          <Text style={styles.name}>
            {profile?.first_name} {profile?.last_name}
          </Text>
          <Text style={styles.email}>{user?.email}</Text>
          <Text style={styles.phone}>{profile?.phone}</Text>
        </View>

        {/* Stats Row */}
        <View style={styles.statsRow}>
          <View style={styles.statItem}>
            <Text style={styles.statValue}>{stats.totalDeliveries}</Text>
            <Text style={styles.statLabel}>Deliveries</Text>
          </View>
          <View style={styles.statDivider} />
          <View style={styles.statItem}>
            <Text style={styles.statValue}>{stats.totalEarnings} MAD</Text>
            <Text style={styles.statLabel}>Total Earned</Text>
          </View>
          <View style={styles.statDivider} />
          <View style={styles.statItem}>
            <Text style={styles.statValue}>★ {stats.rating}</Text>
            <Text style={styles.statLabel}>Rating</Text>
          </View>
        </View>

        {/* Menu Items */}
        <View style={styles.menuSection}>
          <TouchableOpacity style={styles.menuItem}>
            <Ionicons name="document-text-outline" size={24} color={theme.colors.textDark} />
            <Text style={styles.menuText}>Driver Guidelines</Text>
            <Ionicons name="chevron-forward" size={20} color={theme.colors.textLight} />
          </TouchableOpacity>
          
          <TouchableOpacity style={styles.menuItem}>
            <Ionicons name="help-circle-outline" size={24} color={theme.colors.textDark} />
            <Text style={styles.menuText}>Help & Support</Text>
            <Ionicons name="chevron-forward" size={20} color={theme.colors.textLight} />
          </TouchableOpacity>
          
          <TouchableOpacity style={styles.menuItem}>
            <Ionicons name="shield-checkmark-outline" size={24} color={theme.colors.textDark} />
            <Text style={styles.menuText}>Privacy Policy</Text>
            <Ionicons name="chevron-forward" size={20} color={theme.colors.textLight} />
          </TouchableOpacity>
        </View>

        {/* Sign Out Button */}
        <Button
          title="Sign Out"
          variant="danger"
          onPress={handleSignOut}
          style={styles.signOutButton}
        />
      </ScrollView>
    </SafeAreaView>
  );
}

import React, { useEffect, useState } from 'react';
import {
  View,
  Text,
  StyleSheet,
  ScrollView,
  TouchableOpacity,
  Alert,
  ActivityIndicator,
  Linking,
} from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { router, useLocalSearchParams } from 'expo-router';
import { Ionicons } from '@expo/vector-icons';
import { theme } from '../../../src/theme';
import { Button } from '../../../src/components/ui/Button';
import { useAuthStore } from '../../../src/stores/authStore';
import { supabase } from '../../../src/lib/supabase';

interface OrderDetails {
  id: string;
  order_number: string;
  status: string;
  total: number;
  delivery_fee: number;
  platform_service_fee: number;
  delivery_address: string;
  payment_method: string;
  customer_name: string;
  customer_phone: string;
  customer_email: string;
  created_at: string;
  items: any[];
  cash_collected: boolean;
}

export default function DriverOrderDetails() {
  const { id } = useLocalSearchParams<{ id: string }>();
  const { user } = useAuthStore();
  const [order, setOrder] = useState<OrderDetails | null>(null);
  const [loading, setLoading] = useState(true);
  const [updating, setUpdating] = useState(false);

  useEffect(() => {
    fetchOrderDetails();
    subscribeToOrderUpdates();
  }, [id]);

  const fetchOrderDetails = async () => {
    // Get the order
    const { data: orderData, error: orderError } = await supabase
      .from('orders')
      .select(`
        *,
        order_items (*)
      `)
      .eq('id', id)
      .maybeSingle();
    
    if (!orderError && orderData) {
      // Get customer profile
      const { data: profileData } = await supabase
        .from('user_profiles')
        .select('full_name, phone')
        .eq('user_id', orderData.user_id)
        .maybeSingle();
      
      // Check if cash was already collected
      const { data: cashData } = await supabase
        .from('cash_collections')
        .select('id')
        .eq('order_id', orderData.id)
        .eq('driver_id', user?.id)
        .maybeSingle();
      
      setOrder({
        id: orderData.id,
        order_number: orderData.order_number,
        status: orderData.status,
        total: orderData.total,
        delivery_fee: orderData.delivery_fee || 15,
        platform_service_fee: orderData.platform_service_fee || 19,
        delivery_address: orderData.delivery_address,
        payment_method: orderData.payment_method || 'cash',
        customer_name: profileData?.full_name || 'Customer',
        customer_phone: profileData?.phone || '',
        customer_email: '',
        created_at: orderData.created_at,
        items: orderData.order_items || [],
        cash_collected: !!cashData,
      });
    }
    setLoading(false);
  };

  const subscribeToOrderUpdates = () => {
    const subscription = supabase
      .channel(`driver-order-${id}`)
      .on(
        'postgres_changes',
        {
          event: 'UPDATE',
          schema: 'public',
          table: 'orders',
          filter: `id=eq.${id}`,
        },
        (payload) => {
          if (payload.new) {
            setOrder(prev => prev ? { ...prev, status: payload.new.status } : prev);
          }
        }
      )
      .subscribe();

    return () => subscription.unsubscribe();
  };

  const updateOrderStatus = async (newStatus: string) => {
    if (!order || !user) return;
    
    setUpdating(true);
    
    const updateData: any = { status: newStatus };
    if (newStatus === 'picked_up') {
      updateData.picked_up_at = new Date().toISOString();
    } else if (newStatus === 'delivered') {
      updateData.delivered_at = new Date().toISOString();
    }
    
    const { error } = await supabase
      .from('orders')
      .update(updateData)
      .eq('id', order.id);
    
    if (!error) {
      setOrder({ ...order, status: newStatus });
      Alert.alert('Success', `Order status updated to ${newStatus}`);
    } else {
      Alert.alert('Error', 'Failed to update order status');
    }
    
    setUpdating(false);
  };

  const handleCashCollected = async () => {
    if (!order || !user) return;
    
    const deliveryAmount = order.delivery_fee;
    
    Alert.alert(
      'Collect Cash',
      `Confirm you have collected ${deliveryAmount} MAD (delivery fee) from the customer?`,
      [
        { text: 'Cancel', style: 'cancel' },
        {
          text: 'Confirm Collected',
          onPress: async () => {
            setUpdating(true);
            
            // Record cash collection
            const { error } = await supabase
              .from('cash_collections')
              .insert({
                driver_id: user.id,
                order_id: order.id,
                amount: deliveryAmount,
                order_number: order.order_number,
                collected_at: new Date().toISOString(),
              });
            
            if (!error) {
              setOrder({ ...order, cash_collected: true });
              Alert.alert(
                'Cash Collected', 
                `You have collected ${deliveryAmount} MAD.\n\nThis amount will be recorded and you will receive your 15% commission (${Math.round(deliveryAmount * 0.15)} MAD) in your weekly payout.`
              );
              
              // Auto-mark as delivered if not already
              if (order.status !== 'delivered') {
                await updateOrderStatus('delivered');
              }
            } else {
              Alert.alert('Error', 'Failed to record cash collection');
            }
            
            setUpdating(false);
          },
        },
      ]
    );
  };

  const openNavigation = () => {
    const url = `https://www.google.com/maps/dir/?api=1&destination=${encodeURIComponent(order?.delivery_address || '')}`;
    Linking.openURL(url);
  };

  const callCustomer = () => {
    if (order?.customer_phone) {
      Linking.openURL(`tel:${order.customer_phone}`);
    } else {
      Alert.alert('No phone number', 'Customer phone number not available');
    }
  };

  const getStatusSteps = () => {
    const steps = [
      { key: 'accepted', label: 'Accepted', icon: 'checkmark-circle' },
      { key: 'picked_up', label: 'Picked Up', icon: 'restaurant' },
      { key: 'on_the_way', label: 'On The Way', icon: 'bicycle' },
      { key: 'delivered', label: 'Delivered', icon: 'home' },
    ];
    
    const currentIndex = steps.findIndex(s => s.key === order?.status);
    
    return steps.map((step, index) => ({
      ...step,
      completed: index <= currentIndex,
      current: index === currentIndex,
    }));
  };

  if (loading) {
    return (
      <View style={styles.loadingContainer}>
        <ActivityIndicator size="large" color={theme.colors.gold} />
      </View>
    );
  }

  if (!order) {
    return (
      <View style={styles.errorContainer}>
        <Ionicons name="alert-circle" size={64} color={theme.colors.error} />
        <Text style={styles.errorText}>Order not found</Text>
        <Button title="Go Back" onPress={() => router.back()} />
      </View>
    );
  }

  const statusSteps = getStatusSteps();
  const isCashPayment = order.payment_method === 'cash';
  const needsCashCollection = isCashPayment && !order.cash_collected && order.status === 'delivered';
  const commissionAmount = Math.round(order.delivery_fee * 0.15);

  return (
    <SafeAreaView style={styles.container}>
      <View style={styles.header}>
        <TouchableOpacity onPress={() => router.back()} style={styles.backButton}>
          <Ionicons name="arrow-back" size={24} color={theme.colors.textDark} />
        </TouchableOpacity>
        <Text style={styles.title}>Order #{order.order_number}</Text>
        <View style={{ width: 40 }} />
      </View>

      <ScrollView showsVerticalScrollIndicator={false}>
        {/* Status Timeline */}
        <View style={styles.timelineContainer}>
          {statusSteps.map((step, index) => (
            <View key={step.key} style={styles.timelineStep}>
              <View style={[
                styles.timelineIcon,
                step.completed && styles.timelineIconCompleted,
                step.current && styles.timelineIconCurrent,
              ]}>
                <Ionicons
                  name={step.icon as any}
                  size={20}
                  color={step.completed ? theme.colors.gold : theme.colors.textLight}
                />
              </View>
              {index < statusSteps.length - 1 && (
                <View style={[
                  styles.timelineLine,
                  step.completed && styles.timelineLineCompleted,
                ]} />
              )}
              <Text style={[
                styles.timelineLabel,
                step.completed && styles.timelineLabelCompleted,
              ]}>
                {step.label}
              </Text>
            </View>
          ))}
        </View>

        {/* Payment Method Card */}
        <View style={[styles.paymentCard, { backgroundColor: isCashPayment ? '#f59e0b20' : '#10b98120' }]}>
          <Ionicons 
            name={isCashPayment ? "cash-outline" : "card-outline"} 
            size={24} 
            color={isCashPayment ? '#f59e0b' : '#10b981'} 
          />
          <Text style={styles.paymentText}>
            Payment: <Text style={[styles.paymentValue, { color: isCashPayment ? '#f59e0b' : '#10b981' }]}>
              {isCashPayment ? 'Cash on Delivery' : 'Card Payment'}
            </Text>
          </Text>
        </View>

        {/* Cash Collection Status */}
        {isCashPayment && (
          <View style={[styles.cashStatusCard, { backgroundColor: order.cash_collected ? '#10b98120' : '#f59e0b20' }]}>
            <Ionicons 
              name={order.cash_collected ? "checkmark-circle" : "alert-circle"} 
              size={20} 
              color={order.cash_collected ? '#10b981' : '#f59e0b'} 
            />
            <Text style={[styles.cashStatusText, { color: order.cash_collected ? '#10b981' : '#f59e0b' }]}>
              {order.cash_collected ? 'Cash collected ✓' : 'Awaiting cash collection'}
            </Text>
          </View>
        )}

        {/* Customer Info Card */}
        <View style={styles.contactCard}>
          <Text style={styles.contactTitle}>Customer Information</Text>
          <View style={styles.customerRow}>
            <Ionicons name="person-outline" size={18} color={theme.colors.textLight} />
            <Text style={styles.customerText}>{order.customer_name}</Text>
          </View>
          <View style={styles.customerRow}>
            <Ionicons name="call-outline" size={18} color={theme.colors.textLight} />

        </View>

        {/* Map Preview */}
        <View style={styles.mapContainer}>
          <View style={styles.mapFallback}>

              {order.delivery_address}

          <TouchableOpacity style={styles.navigateButton} onPress={openNavigation}>
            <Ionicons name="navigate" size={20} color={theme.colors.textWhite} />
            <Text style={styles.navigateText}>Open in Google Maps</Text>
          </TouchableOpacity>
        </View>

        {/* Order Items */}
        <View style={styles.infoCard}>
          <Text style={styles.infoTitle}>Order Items</Text>

          ))}
          
          <View style={styles.infoDivider} />

        {/* Contact Customer Button */}

            <Text style={styles.callButtonText}>Contact Customer</Text>

        </View>

        {/* Action Buttons */}
        <View style={styles.actionContainer}>
          {order.status === 'accepted' && (
            <Button
              title="Picked Up Order"
